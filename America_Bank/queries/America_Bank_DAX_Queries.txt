----------------------------------------------------------------
Query 1: Quarter range (Min/Max)
----------------------------------------------------------------
EVALUATE
	ROW(
		"MinQuarter", CALCULATE(MIN('America Bank New'[Quarter])),
		"MaxQuarter", CALCULATE(MAX('America Bank New'[Quarter]))
	)

----------------------------------------------------------------
Query 2: Distinct list of Contacts (top 100)
----------------------------------------------------------------
DEFINE
	VAR __DS0Core = 
		VALUES('America Bank New'[contact])
	VAR __DS0PrimaryWindowed = 
		TOPN(101, __DS0Core, 'America Bank New'[contact], 1)
EVALUATE
	__DS0PrimaryWindowed
ORDER BY
	'America Bank New'[contact]

----------------------------------------------------------------
Query 3: Customer count by Job Classification
----------------------------------------------------------------
DEFINE
	VAR __DS0Core = 
		SUMMARIZECOLUMNS(
			'America Bank New'[Job Classification],
			"CountCustomer_ID", CALCULATE(COUNTA('America Bank New'[Customer ID]))
		)
	VAR __DS0BodyLimited = 
		TOPN(1002, __DS0Core, [CountCustomer_ID], 0, 'America Bank New'[Job Classification], 1)
EVALUATE
	__DS0BodyLimited
ORDER BY
	[CountCustomer_ID] DESC, 'America Bank New'[Job Classification]

----------------------------------------------------------------
Query 4: Total Customer Count (single value)

	)

Query 10: Customer count by State (with Min/Max group summary)
----------------------------------------------------------------

			'America Bank New'[State],

----------------------------------------------------------------
Query 10b: State Min/Max group summary (run separately if needed)
----------------------------------------------------------------
DEFINE
	VAR __DS0Core = 
		SUMMARIZECOLUMNS(
			'America Bank New'[State],
			"CountState", CALCULATE(COUNTA('America Bank New'[State]))
		)
EVALUATE
	GROUPBY(
		__DS0Core,
		"MinCountState", MINX(CURRENTGROUP(), [CountState]),
		"MaxCountState", MAXX(CURRENTGROUP(), [CountState])
	)
